                const targetElement = document.querySelector(this.getAttribute('href'));
                if(targetElement) {
                    // Adjust for fixed header height
                    const headerOffset = document.getElementById('header').offsetHeight;
                    const elementPosition = targetElement.getBoundingClientRect().top;
                    const offsetPosition = elementPosition + window.pageYOffset - headerOffset;

                    window.scrollTo({
                        top: offsetPosition,
                        behavior: "smooth"
                    });
                }
            });
        });

        // Auto-scrolling carousels
        const scrollers = document.querySelectorAll(".scroller");
        if (!window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
            addAnimation();
        }

        function addAnimation() {
            scrollers.forEach((scroller) => {
                scroller.setAttribute("data-animated", true);
            });
        }

        // --- Modal Logic ---
        function setupModal(triggerSelector, modalId) {
            const triggers = document.querySelectorAll(triggerSelector);
            const modal = document.getElementById(modalId);
            if (!triggers.length || !modal) return;

            const closeBtns = modal.querySelectorAll('.modal-close-btn');
            const yesBtnJoin = modal.querySelector('.modal-yes-btn-join');
            const yesBtnWork = modal.querySelector('.modal-yes-btn-work');
            const noBtns = modal.querySelectorAll('.modal-no-btn');
            const initialView = modal.querySelector('.modal-initial-view');
            const noView = modal.querySelector('.modal-no-view');
            const teamsView = modal.querySelector('.modal-teams-view');
            const teamLinks = modal.querySelectorAll('.team-gform-link');

            const showModal = () => {
                initialView.classList.remove('hidden');
                noView.classList.add('hidden');
                if (teamsView) teamsView.classList.add('hidden');
                modal.classList.remove('hidden');
            };

            const hideModal = () => {
                modal.classList.add('hidden');
            };

            triggers.forEach(button => {
                button.addEventListener('click', (e) => {
                    e.preventDefault();
                    showModal();
                });
            });

            closeBtns.forEach(btn => btn.addEventListener('click', hideModal));

            modal.addEventListener('click', (e) => {
                if (e.target === modal) {
                    hideModal();
                }
            });

            noBtns.forEach(btn => {
                btn.addEventListener('click', () => {
                    initialView.classList.add('hidden');
                    if (teamsView) teamsView.classList.add('hidden');
                    noView.classList.remove('hidden');
                });
            });

            // "Join Us" modal logic
            if (yesBtnJoin) {
                yesBtnJoin.addEventListener('click', () => {
                    // IMPORTANT: Replace with your actual WhatsApp community link
                    window.open('https://chat.whatsapp.com/JmxtsZbjJLpH1Q0O8flN6L?mode=ac_t', '_blank');
                    hideModal();
                });
            }

            // "Work With Us" modal logic
            if (yesBtnWork) {
                yesBtnWork.addEventListener('click', () => {
                    initialView.classList.add('hidden');
                    teamsView.classList.remove('hidden');
                });
            }

            if (teamLinks) {
                teamLinks.forEach(link => {
                    link.addEventListener('click', () => {
                        // IMPORTANT: Replace with your actual Google Form link
                        window.open('https://forms.gle/813Rve6DH5yUASqM9', '_blank');
                        hideModal();
                    });
                });
            }
        }

        setupModal('.join-trigger', 'join-modal');
        setupModal('.work-trigger', 'work-modal');

    </script>
